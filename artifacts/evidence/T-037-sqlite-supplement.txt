import java.lang.reflect.Field;
import java.util.List;

/** Temporary probe: prints the real DDL strings the production schema builds, so the desktop
 *  SQLite supplement executes the shipped text instead of a hand-copied second edition. */
public final class DumpSchema {

    public static void main(String[] args) throws Exception {
        Class<?> schema = Class.forName("com.treelearn.data.db.TreeLearnSchema");
        Class<?> messages = Class.forName("com.treelearn.data.db.TreeLearnSchema$Messages");
        scalar("DATABASE_VERSION", schema.getField("DATABASE_VERSION").get(null));
        scalar("QUOTED_TEXT", messages.getField("QUOTED_TEXT").get(null));
        scalar("QUOTED_TEXT_DECLARATION", messages.getField("QUOTED_TEXT_DECLARATION").get(null));
        list("CREATE_TABLE_STATEMENTS", schema.getField("CREATE_TABLE_STATEMENTS"));
        list("CREATE_INDEX_STATEMENTS", schema.getField("CREATE_INDEX_STATEMENTS"));
        list("CREATE_TRIGGER_STATEMENTS", schema.getField("CREATE_TRIGGER_STATEMENTS"));
        list("MESSAGES_QUOTE_INVARIANT_STATEMENTS",
                schema.getField("MESSAGES_QUOTE_INVARIANT_STATEMENTS"));
        list("MIGRATION_V1_TO_V2_STATEMENTS",
                schema.getField("MIGRATION_V1_TO_V2_STATEMENTS"));
    }

    private static void scalar(String name, Object value) {
        System.out.println("@@SCALAR@@" + name + "@@" + value);
    }

    @SuppressWarnings("unchecked")
    private static void list(String name, Field field) throws Exception {
        for (String statement : (List<String>) field.get(null)) {
            System.out.println("@@STMT@@" + name + "@@" + statement);
        }
        System.out.println("@@END@@" + name);
    }
}

"""T-037 supplementary probe (NOT device evidence).

Executes the DDL text the production schema builds (dumped from the compiled
com.treelearn.data.db.TreeLearnSchema constants by DumpSchema.java, so this is the shipped
text and not a hand-copied second edition) on a real SQLite engine, on BOTH install paths:

  fresh   : CREATE_TABLE + CREATE_INDEX + CREATE_TRIGGER              (onCreate)
  upgraded: V1 shape + MIGRATION_V1_TO_V2_STATEMENTS                  (onUpgrade)

and compares the two paths input by input.

Desktop SQLite is a supplement, never a substitute for API 28+ Android SQLite (rules/sql.md);
it is recorded as such. The device rounds were abandoned by the user on 2026-09-21.
"""
import hashlib
import os
import sqlite3
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
DUMP = os.path.join(HERE, "ddl-dump.txt")

SECTIONS = ("CREATE_TABLE_STATEMENTS", "CREATE_INDEX_STATEMENTS", "CREATE_TRIGGER_STATEMENTS",
            "MESSAGES_QUOTE_INVARIANT_STATEMENTS", "MIGRATION_V1_TO_V2_STATEMENTS")


def load(path):
    scalars, statements = {}, {name: [] for name in SECTIONS}
    with open(path, "r", encoding="utf-8") as handle:
        for raw in handle:
            line = raw.rstrip("\n")
            if line.startswith("@@SCALAR@@"):
                name, value = line[len("@@SCALAR@@"):].split("@@", 1)
                scalars[name] = value
            elif line.startswith("@@STMT@@"):
                name, value = line[len("@@STMT@@"):].split("@@", 1)
                statements[name].append(value)
    return scalars, statements


def v1_shape(statements, declaration):
    """The V1 database: the same DDL minus the V2 column and minus the V2 triggers.

    Derived mechanically from the shipped text so V1 differs from V2 in exactly one column.
    """
    drop = ", " + declaration
    tables = []
    for statement in statements["CREATE_TABLE_STATEMENTS"]:
        if statement.startswith("CREATE TABLE messages "):
            assert statement.count(drop) == 1, "expected exactly one declaration to remove"
            statement = statement.replace(drop, "")
        tables.append(statement)
    invariant = set(statements["MESSAGES_QUOTE_INVARIANT_STATEMENTS"])
    triggers = [s for s in statements["CREATE_TRIGGER_STATEMENTS"] if s not in invariant]
    return tables, statements["CREATE_INDEX_STATEMENTS"], triggers


def build(stmts):
    conn = sqlite3.connect(":memory:")
    conn.execute("PRAGMA foreign_keys=ON")
    for statement in stmts:
        conn.execute(statement)
    return conn


def columns(conn, table):
    return [row[1] for row in conn.execute("PRAGMA table_info(" + table + ")")]


def scenario(conn):
    """Every input is a state the domain layer can ask the store to hold. Each probe violates at
    most one rule."""
    results = []
    cursor = conn.cursor()
    cursor.execute("INSERT INTO topics (id, created_at, updated_at) VALUES ('topic-1', 1, 1)")
    cursor.execute("INSERT INTO branches (id, topic_id, created_at, updated_at) VALUES"
                   " ('branch-1', 'topic-1', 1, 1)")

    def insert(case, ordinal, role, content, quoted):
        try:
            cursor.execute("INSERT INTO messages (id, branch_id, ordinal, role, user_content,"
                           " created_at, quoted_text) VALUES (?, 'branch-1', ?, ?, ?, 1, ?)",
                           ("m-" + case, ordinal, role, content, quoted))
            return {"case": case, "outcome": "accepted", "code": None}
        except sqlite3.IntegrityError as exc:
            return {"case": case, "outcome": "refused",
                    "code": getattr(exc, "sqlite_errorcode", None)}

    results.append(insert("C1-user-with-quote", 0, "USER", "q1", "the quoted sentence"))
    results.append(insert("C2-user-null-quote", 1, "USER", "q2", None))
    results.append(insert("C3-user-empty-quote", 2, "USER", "q3", ""))
    results.append(insert("C4-ai-with-quote", 3, "AI", None, "the quoted sentence"))
    results.append(insert("C5-ai-null-quote", 4, "AI", None, None))

    # C6 drives the UPDATE half: the invariant has to hold on both write events.
    try:
        cursor.execute("UPDATE messages SET quoted_text='x' WHERE id='m-C5-ai-null-quote'")
        results.append({"case": "C6-update-ai-sets-quote", "outcome": "accepted", "code": None})
    except sqlite3.IntegrityError as exc:
        results.append({"case": "C6-update-ai-sets-quote", "outcome": "refused",
                        "code": getattr(exc, "sqlite_errorcode", None)})

    # Read-back: what the DAO would see through the cursor.
    read = cursor.execute("SELECT quoted_text FROM messages WHERE id='m-C1-user-with-quote'") \
        .fetchone()[0]
    results.append({"case": "C7-readback-user-quote", "outcome": read, "code": None})
    read = cursor.execute("SELECT quoted_text FROM messages WHERE id='m-C2-user-null-quote'") \
        .fetchone()[0]
    results.append({"case": "C8-readback-null", "outcome": read, "code": None})
    return results


def render(rows):
    return ["  %-26s %-10s %s" % (r["case"], r["outcome"],
                                  "" if r["code"] is None else str(r["code"])) for r in rows]


def main():
    scalars, statements = load(DUMP)
    declaration = scalars["QUOTED_TEXT_DECLARATION"]
    print("DDL source  : %s" % DUMP)
    print("DDL sha256  : %s" % hashlib.sha256(open(DUMP, "rb").read()).hexdigest())
    print("SQLite      : %s" % sqlite3.sqlite_version)
    print("DATABASE_VERSION=%s  QUOTED_TEXT=%s  DECLARATION=%s"
          % (scalars["DATABASE_VERSION"], scalars["QUOTED_TEXT"], declaration))

    fresh = build(statements["CREATE_TABLE_STATEMENTS"]
                  + statements["CREATE_INDEX_STATEMENTS"]
                  + statements["CREATE_TRIGGER_STATEMENTS"])

    v1_tables, v1_indexes, v1_triggers = v1_shape(statements, declaration)
    upgraded = build(v1_tables + v1_indexes + v1_triggers)
    print("\nupgraded before migration, messages columns: %s" % columns(upgraded, "messages"))
    assert scalars["QUOTED_TEXT"] not in columns(upgraded, "messages"), "V1 must not have the column"
    for statement in statements["MIGRATION_V1_TO_V2_STATEMENTS"]:
        upgraded.execute(statement)

    fresh_columns = columns(fresh, "messages")
    upgraded_columns = columns(upgraded, "messages")
    print("fresh    messages columns: %s" % fresh_columns)
    print("upgraded messages columns: %s" % upgraded_columns)
    print("column shapes identical  : %s" % (fresh_columns == upgraded_columns))

    print("\nfresh install (onCreate):")
    fresh_rows = scenario(fresh)
    print("\n".join(render(fresh_rows)))
    print("\nupgraded install (onUpgrade -> V1->V2):")
    upgraded_rows = scenario(upgraded)
    print("\n".join(render(upgraded_rows)))

    print("\nOK: no device evidence is claimed here; this is a desktop-SQLite supplement only.")
    failures = []
    if fresh_columns != upgraded_columns:
        failures.append("column shapes differ")
    if fresh_rows != upgraded_rows:
        failures.append("the two install paths disagree on an input")
    for row in fresh_rows:
        if row["case"].startswith(("C3", "C4", "C6")) and row["code"] != 1811:
            failures.append("%s expected extended code 1811, got %s" % (row["case"], row["code"]))
    if [r for r in fresh_rows if r["case"] == "C8-readback-null"][0]["outcome"] is not None:
        failures.append("a NULL quote must read back as SQL NULL to the DAO")
    if failures:
        print("\nSUPPLEMENT FAIL: " + "; ".join(failures))
        return 1
    print("SUPPLEMENT PASS: both install paths accept and refuse the same inputs, and every"
          " refusal carries extended code 1811 (SQLITE_CONSTRAINT_TRIGGER).")
    return 0


if __name__ == "__main__":
    sys.exit(main())
