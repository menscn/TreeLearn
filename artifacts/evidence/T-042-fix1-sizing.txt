def strip(text, xml=False):
    if xml:
        return re.sub(r"<!--.*?-->", "", text, flags=re.S)
    text = re.sub(r"/\*.*?\*/", "", text, flags=re.S)
    return re.sub(r"//[^\n]*", "", text)

def count(text, xml=False):
    return len([l for l in strip(text, xml).splitlines() if l.strip()])

def read(path):
    with open(path, encoding="utf-8") as h:
        return h.read()

def member(text, sig):
    """brace-matched Java member starting at the line that contains sig"""
    idx = text.find(sig)
    if idx < 0:
        return None
    start = text.rfind("\n", 0, idx) + 1  # the signature line itself
    brace = text.find("{", idx)
    depth = 0
    i = brace
    while i < len(text):
        if text[i] == "{":
            depth += 1
        elif text[i] == "}":
            depth -= 1
            if depth == 0:
                return text[start:i + 1]
        i += 1
    return None

def xml_element(text, marker):
    idx = text.find(marker)
    if idx < 0:
        return None
    start = text.rfind("<TextView", 0, idx)
    end = text.find("/>", idx)
    if start < 0 or end < 0:
        return None
    return text[start:end + 2]
