def is_prose(s):
    t = s.strip()
    if not t:
        return True
    return t.startswith("//") or t.startswith("/*") or t.startswith("*") or t.startswith("<!--")

def count_judgement(path):
    lines = open(path, encoding="utf-8").read().splitlines()
    n = 0
    closer = None
    for ln in lines:
        s = ln.strip()
        if closer is not None:
            if closer in s:
                closer = None
            continue
        if s.startswith("<!--"):
            if "-->" not in s:
                closer = "-->"
            continue
        if s.startswith("/*"):
            if "*/" not in s:
                closer = "*/"
            continue
        if is_prose(ln):
            continue
        n += 1
    return n, len(lines)
