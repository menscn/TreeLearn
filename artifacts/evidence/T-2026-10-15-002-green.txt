    即修复前的形状——写在 try 之外，中间崩溃就把残留留在被跟踪文件里。"""
    lines = src_text.splitlines()
    hits = [i for i, l in enumerate(lines) if l.strip().startswith(INJECT_MARK)]
    if len(hits) != 1:
        return None
    i = hits[0]
    l = lines[i]
    if not l.startswith("    "):
        return None
    out = list(lines)
    out[i] = l[4:]
    return "\n".join(out)
